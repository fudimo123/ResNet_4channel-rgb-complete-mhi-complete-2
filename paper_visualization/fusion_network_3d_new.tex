% Draw a 3D Block (Cube/Cuboid)
% #1: x, #2: y, #3: z (center)
% #4: width (x), #5: depth (y), #6: height (z)
% #7: color, #8: label, #9: opacity
\newcommand{\DrawBlock}[9]{
    % Back face (invisible mostly, but good for completeness)
    \fill[#7, opacity=#9] (#1+#4/2, #2-#5/2, #3-#6/2) -- (#1-#4/2, #2-#5/2, #3-#6/2) -- (#1-#4/2, #2+#5/2, #3-#6/2) -- (#1+#4/2, #2+#5/2, #3-#6/2) -- cycle;
    % Bottom face
    \fill[#7, opacity=#9] (#1+#4/2, #2-#5/2, #3-#6/2) -- (#1-#4/2, #2-#5/2, #3-#6/2) -- (#1-#4/2, #2-#5/2, #3+#6/2) -- (#1+#4/2, #2-#5/2, #3+#6/2) -- cycle;
    % Left face
    \fill[#7, opacity=#9] (#1-#4/2, #2-#5/2, #3-#6/2) -- (#1-#4/2, #2+#5/2, #3-#6/2) -- (#1-#4/2, #2+#5/2, #3+#6/2) -- (#1-#4/2, #2-#5/2, #3+#6/2) -- cycle;
    
    % Top face (Visible)
    \filldraw[fill=#7!80!white, draw=#7!40!black] (#1-#4/2, #2+#5/2, #3+#6/2) -- (#1+#4/2, #2+#5/2, #3+#6/2) -- (#1+#4/2, #2-#5/2, #3+#6/2) -- (#1-#4/2, #2-#5/2, #3+#6/2) -- cycle;
    % Front face (Visible)
    \filldraw[fill=#7, draw=#7!40!black] (#1-#4/2, #2-#5/2, #3+#6/2) -- (#1+#4/2, #2-#5/2, #3+#6/2) -- (#1+#4/2, #2-#5/2, #3-#6/2) -- (#1-#4/2, #2-#5/2, #3-#6/2) -- cycle;
    % Right face (Visible)
    \filldraw[fill=#7!60!black, draw=#7!40!black] (#1+#4/2, #2-#5/2, #3+#6/2) -- (#1+#4/2, #2+#5/2, #3+#6/2) -- (#1+#4/2, #2+#5/2, #3-#6/2) -- (#1+#4/2, #2-#5/2, #3-#6/2) -- cycle;
    
    % Label
    \node[anchor=north, align=center, yshift=-0.2cm] at (#1, #2-#5/2, #3-#6/2) {#8};
}

% Draw a 3D Trapezoid (Backbone)
% #1: x_start, #2: x_end, #3: z_center
% #4: height_start, #5: height_end, #6: depth
% #7: color, #8: label
\newcommand{\DrawTrapezoid}[8]{
    \def\xs{#1} \def\xe{#2} \def\zc{#3}
    \def\hs{#4} \def\he{#5} \def\d{#6}
    \def\col{#7}
    
    % Coordinates for front face
    \coordinate (Fbl) at (\xs, -\d/2, \zc-\hs/2);
    \coordinate (Ftl) at (\xs, -\d/2, \zc+\hs/2);
    \coordinate (Fbr) at (\xe, -\d/2, \zc-\he/2);
    \coordinate (Ftr) at (\xe, -\d/2, \zc+\he/2);
    
    % Coordinates for back face
    \coordinate (Bbl) at (\xs, \d/2, \zc-\hs/2);
    \coordinate (Btl) at (\xs, \d/2, \zc+\hs/2);
    \coordinate (Bbr) at (\xe, \d/2, \zc-\he/2);
    \coordinate (Btr) at (\xe, \d/2, \zc+\he/2);
    
    % Top face
    \filldraw[fill=\col!80!white, draw=\col!40!black] (Ftl) -- (Ftr) -- (Btr) -- (Btl) -- cycle;
    % Front face
    \filldraw[fill=\col, draw=\col!40!black] (Fbl) -- (Fbr) -- (Ftr) -- (Ftl) -- cycle;
    % Right face
    \filldraw[fill=\col!60!black, draw=\col!40!black] (Fbr) -- (Bbr) -- (Btr) -- (Ftr) -- cycle;
    
    % Label on front face
    \node[align=center, font=\bfseries] at ($(\xs, -\d/2, \zc)!0.5!(\xe, -\d/2, \zc)$) {#8};
}
