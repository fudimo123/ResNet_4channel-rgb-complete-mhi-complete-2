% --- 3D Trapezoid Macro ---
% #1: x_center, #2: y_center, #3: z_center
% #4: width (x_len), #5: height_left, #6: height_right, #7: depth (z_len)
% #8: color, #9: label
\newcommand{\DrawTrapezoid}[9]{
    \coordinate (C) at (#1, #2, #3);
    \def\w{#4} \def\hl{#5} \def\hr{#6} \def\d{#7} \def\col{#8}
    
    % Front Face (z = -d/2)
    \coordinate (Ftl) at ($(C) + (-\w/2, \hl/2, -\d/2)$);
    \coordinate (Fbl) at ($(C) + (-\w/2, -\hl/2, -\d/2)$);
    \coordinate (Ftr) at ($(C) + (\w/2, \hr/2, -\d/2)$);
    \coordinate (Fbr) at ($(C) + (\w/2, -\hr/2, -\d/2)$);
    
    % Back Face (z = d/2)
    \coordinate (Btl) at ($(C) + (-\w/2, \hl/2, \d/2)$);
    \coordinate (Bbl) at ($(C) + (-\w/2, -\hl/2, \d/2)$);
    \coordinate (Btr) at ($(C) + (\w/2, \hr/2, \d/2)$);
    \coordinate (Bbr) at ($(C) + (\w/2, -\hr/2, \d/2)$);
    
    % Draw Right Face
    \filldraw[fill=\col!60!black, draw=\col!40!black, thick] (Fbr) -- (Bbr) -- (Btr) -- (Ftr) -- cycle;
    % Draw Top Face
    \filldraw[fill=\col!80!white, draw=\col!40!black, thick] (Ftl) -- (Btl) -- (Btr) -- (Ftr) -- cycle;
    % Draw Front Face
    \filldraw[fill=\col, draw=\col!40!black, thick] (Fbl) -- (Fbr) -- (Ftr) -- (Ftl) -- cycle;
    
    \node[align=center, text=white, font=\bfseries] at (C) {#9};
}

% --- 3D Block (Cube) Macro ---
% #1: x, #2: y, #3: z (center)
% #4: width (x), #5: height (y), #6: depth (z)
% #7: color, #8: label, #9: opacity
\newcommand{\DrawBlock}[9]{
    \coordinate (C) at (#1, #2, #3);
    \def\w{#4} \def\h{#5} \def\d{#6} \def\col{#7} \def\op{#9}
    
    % Corners
    \coordinate (Ftl) at ($(C) + (-\w/2, \h/2, -\d/2)$);
    \coordinate (Fbl) at ($(C) + (-\w/2, -\h/2, -\d/2)$);
    \coordinate (Ftr) at ($(C) + (\w/2, \h/2, -\d/2)$);
    \coordinate (Fbr) at ($(C) + (\w/2, -\h/2, -\d/2)$);
    
    \coordinate (Btl) at ($(C) + (-\w/2, \h/2, \d/2)$);
    \coordinate (Bbl) at ($(C) + (-\w/2, -\h/2, \d/2)$);
    \coordinate (Btr) at ($(C) + (\w/2, \h/2, \d/2)$);
    \coordinate (Bbr) at ($(C) + (\w/2, -\h/2, \d/2)$);
    
    % Right Face
    \filldraw[fill=\col!60!black, draw=\col!40!black, opacity=\op] (Fbr) -- (Bbr) -- (Btr) -- (Ftr) -- cycle;
    % Top Face
    \filldraw[fill=\col!80!white, draw=\col!40!black, opacity=\op] (Ftl) -- (Btl) -- (Btr) -- (Ftr) -- cycle;
    % Front Face
    \filldraw[fill=\col, draw=\col!40!black, opacity=\op] (Fbl) -- (Fbr) -- (Ftr) -- (Ftl) -- cycle;
    
    \node[anchor=north, align=center, yshift=-0.2cm] at ($(C) + (0, -\h/2, -\d/2)$) {#8};
}
