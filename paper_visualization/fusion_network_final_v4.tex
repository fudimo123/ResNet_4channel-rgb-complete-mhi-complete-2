% --- MACROS ---
% Trapezoid: #1=x, #2=y, #3=z, #4=width, #5=h_left, #6=h_right, #7=depth, #8=color, #9=text
\newcommand{\DrawTrapezoid}[9]{{
    \coordinate (C) at (#1, #2, #3);
    \def\w{#4} \def\hl{#5} \def\hr{#6} \def\d{#7} 
    \colorlet{tempcol}{#8} % Use colorlet to handle color expressions safely
    % Front Face (z = -d/2)
    \coordinate (Ftl) at ($(C) + (-\w/2, \hl/2, -\d/2)$);
    \coordinate (Fbl) at ($(C) + (-\w/2, -\hl/2, -\d/2)$);
    \coordinate (Ftr) at ($(C) + (\w/2, \hr/2, -\d/2)$);
    \coordinate (Fbr) at ($(C) + (\w/2, -\hr/2, -\d/2)$);
    % Back Face (z = d/2)
    \coordinate (Btl) at ($(C) + (-\w/2, \hl/2, \d/2)$);
    \coordinate (Bbl) at ($(C) + (-\w/2, -\hl/2, \d/2)$);
    \coordinate (Btr) at ($(C) + (\w/2, \hr/2, \d/2)$);
    \coordinate (Bbr) at ($(C) + (\w/2, -\hr/2, \d/2)$);
    % Draw Right Face
    \filldraw[fill=tempcol!60!black, draw=tempcol!40!black, thick] (Fbr) -- (Bbr) -- (Btr) -- (Ftr) -- cycle;
    % Draw Top Face
    \filldraw[fill=tempcol!80!white, draw=tempcol!40!black, thick] (Ftl) -- (Btl) -- (Btr) -- (Ftr) -- cycle;
    % Draw Front Face
    \filldraw[fill=tempcol, draw=tempcol!40!black, thick] (Fbl) -- (Fbr) -- (Ftr) -- (Ftl) -- cycle;
    \node[align=center, text=white, font=\bfseries] at (C) {#9};
}}

% Block/Cuboid: #1=x, #2=y, #3=z, #4=w, #5=h, #6=d, #7=color, #8=text, #9=opacity
\newcommand{\DrawBlock}[9]{{
    \coordinate (C) at (#1, #2, #3);
    \def\w{#4} \def\h{#5} \def\d{#6} \def\op{#9}
    \colorlet{tempcol}{#7} % Use colorlet to handle color expressions safely
    \coordinate (Ftl) at ($(C) + (-\w/2, \h/2, -\d/2)$);
    \coordinate (Fbl) at ($(C) + (-\w/2, -\h/2, -\d/2)$);
    \coordinate (Ftr) at ($(C) + (\w/2, \h/2, -\d/2)$);
    \coordinate (Fbr) at ($(C) + (\w/2, -\h/2, -\d/2)$);
    \coordinate (Btl) at ($(C) + (-\w/2, \h/2, \d/2)$);
    \coordinate (Bbl) at ($(C) + (-\w/2, -\h/2, \d/2)$);
    \coordinate (Btr) at ($(C) + (\w/2, \h/2, \d/2)$);
    \coordinate (Bbr) at ($(C) + (\w/2, -\h/2, \d/2)$);
    \filldraw[fill=tempcol!60!black, draw=tempcol!40!black, opacity=\op] (Fbr) -- (Bbr) -- (Btr) -- (Ftr) -- cycle;
    \filldraw[fill=tempcol!80!white, draw=tempcol!40!black, opacity=\op] (Ftl) -- (Btl) -- (Btr) -- (Ftr) -- cycle;
    \filldraw[fill=tempcol, draw=tempcol!40!black, opacity=\op] (Fbl) -- (Fbr) -- (Ftr) -- (Ftl) -- cycle;
    \node[anchor=north, align=center, yshift=-0.2cm] at ($(C) + (0, -\h/2, -\d/2)$) {#8};
}}
